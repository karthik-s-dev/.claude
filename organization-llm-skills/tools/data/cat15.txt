- UNRESOLVED: Node 20 vs 22 (C-20) - follow the repo; propose alignment via ADR.
impl:
- Start from the template; build and run locally with a dummy token; verify image has no `.npmrc` and runs as non-root.
ex: dockerfile | Runtime stage (reference)
FROM node:22-alpine AS production
RUN apk add --no-cache dumb-init && addgroup -g 1001 -S nodejs && adduser -S nodejs -u 1001
WORKDIR /app
ENV NODE_ENV=production
COPY --from=builder /app/package*.json ./
COPY --from=builder /app/node_modules ./node_modules
COPY --from=builder /app/dist ./dist
RUN chown -R nodejs:nodejs /app
USER nodejs
EXPOSE 3001
ENTRYPOINT ["dumb-init", "--"]
CMD ["node", "dist/index.js"]
/ex
